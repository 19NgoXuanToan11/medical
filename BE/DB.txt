-- Enhanced Database Schema - Report-Dashboard Integration
-- Connecting existing tables without adding new ones
Create Database Medical
Use Medical

CREATE TABLE [Staff] (
  [StaffID] int IDENTITY(1,1) PRIMARY KEY,
  [Username] nvarchar(50) UNIQUE NOT NULL,
  [PasswordHash] nvarchar(255) NOT NULL,
  [Email] nvarchar(100) UNIQUE NOT NULL,
  [FirstName] nvarchar(50) NOT NULL,
  [LastName] nvarchar(50) NOT NULL,
  [Phone] nvarchar(20),
  [RoleID] int NOT NULL, 
);

CREATE TABLE [Student] (
  [StudentID] int IDENTITY(1,1) PRIMARY KEY,
  [StaffID] int,
  [StudentCode] nvarchar(20) UNIQUE NOT NULL,
  [FirstName] nvarchar(50) NOT NULL,
  [LastName] nvarchar(50) NOT NULL,
  [DateOfBirth] date NOT NULL,
  [Gender] char(1) NOT NULL,
  [ClassName] nvarchar(50) NOT NULL,
  [GradeLevel] int NOT NULL,
  [Address] nvarchar(255),
  [IsActive] bit DEFAULT (1)
);

CREATE TABLE [Parent] (
  [ParentID] int IDENTITY(1,1) PRIMARY KEY,
  [StaffID] int,
  [StudentID] int,
  [FirstName] nvarchar(50) NOT NULL,
  [LastName] nvarchar(50) NOT NULL,
  [Relationship] varchar(20) NOT NULL,
  [Phone] nvarchar(20) NOT NULL,
  [Email] nvarchar(100),
  [Address] nvarchar(255),
  [Occupation] nvarchar(100),
  [IsEmergencyContact] bit DEFAULT (0),
  [IsMainContact] bit DEFAULT (0),
  [IsActive] bit DEFAULT (1)
);

CREATE TABLE [Health_Profile] (
  [HealthProfileID] int IDENTITY(1,1) PRIMARY KEY,
  [StudentID] int,
  [BloodType] varchar(5),
  [Height] decimal(5,2),
  [Weight] decimal(5,2),
  [HasAllergies] bit DEFAULT (0),
  [AllergyDetails] nvarchar(1000),
  [HasChronicDiseases] bit DEFAULT (0),
  [ChronicDetails] nvarchar(1000),
  [HasPreviousTreatment] bit DEFAULT (0),
  [TreatmentDetails] nvarchar(1000),
  [HasCompleteVaccinations] varchar(10),
  [VaccinationDetails] nvarchar(1000),
  [Vaccinations] nvarchar(1000),
  [HasVisionIssues] bit DEFAULT (0),
  [LeftEye] nvarchar(20),
  [RightEye] nvarchar(20),
  [VisionNotes] nvarchar(1000),
  [HasHearingIssues] bit DEFAULT (0),
  [LeftEar] nvarchar(100),
  [RightEar] nvarchar(100),
  [HearingNotes] nvarchar(1000),
  [EmergencyContact] nvarchar(255),
  [OtherInfo] nvarchar(1000),
  [LastUpdated] datetime DEFAULT (getdate())
);

CREATE TABLE [Health_Event] (
  [EventID] int IDENTITY(1,1) PRIMARY KEY,
  [StudentID] int,
  [StaffID] int,
  [EventDate] datetime NOT NULL DEFAULT (getdate()),
  [EventType] varchar(50) NOT NULL,
  [Symptoms] nvarchar(500),
  [Assessment] nvarchar(1000),
  [Treatment] nvarchar(1000),
  [ParentNotified] bit DEFAULT (0),
  [FollowUpRequired] bit DEFAULT (0),
  [Notes] nvarchar(500),
  [MedicinesUsed] nvarchar(500),
  [SuppliesUsed] nvarchar(500)
);

CREATE TABLE [Medicine] (
  [MedicineID] int IDENTITY(1,1) PRIMARY KEY,
  [Name] nvarchar(100) NOT NULL,
  [StockQuantity] decimal(10,2) DEFAULT (0),
  [IsActive] bit DEFAULT (1)
);

CREATE TABLE [Medical_Supply] (
  [SupplyID] int IDENTITY(1,1) PRIMARY KEY,
  [Name] nvarchar(100) NOT NULL,
  [Category] varchar(50) NOT NULL,
  [Description] nvarchar(255),
  [StockQuantity] decimal(10,2) DEFAULT (0),
  [IsActive] bit DEFAULT (1)
);

CREATE TABLE [Medicine_Request] (
  [RequestID] int IDENTITY(1,1) PRIMARY KEY,
  [StudentID] int,
  [MedicineName] nvarchar(100) NOT NULL,
  [Dosage] nvarchar(100) NOT NULL,
  [Frequency] nvarchar(100) NOT NULL,
  [TimeOfDay] nvarchar(100),
  [MealRelation] varchar(50),
  [StartDate] date NOT NULL,
  [EndDate] date,
  [Instructions] nvarchar(500),
  [MedicationImagePath] nvarchar(255),
  [PrescriptionImagePath] nvarchar(255),
  [Status] varchar(20) DEFAULT 'Pending',
  [RequestDate] datetime DEFAULT (getdate()),
  [ParentID] int NULL,
  [StaffID] int NULL
);

CREATE TABLE [Request_Result] (
  [ResultID] int IDENTITY(1,1) PRIMARY KEY,
  [RequestID] int,
  [Status] varchar(20) NOT NULL,
  [SubmittedAt] datetime DEFAULT (getdate()),
  [AdministeredBy] int NULL,
  [AdministeredTime] datetime NULL,
  [ActionBy] int NULL,
);

CREATE TABLE [Injection_Form] (
  [FormID] int IDENTITY(1,1) PRIMARY KEY,
  [StudentID] int,
  [ParentID] int,
  [CreatedDate] datetime DEFAULT (getdate()),
  [InjectionName] nvarchar(100) NOT NULL,
  [Description] nvarchar(500),
  [ConsentStatus] varchar(20) DEFAULT 'Pending',
  [ConsentDate] datetime,
  [GradeLevel] int NULL,
  [ClassName] nvarchar(50) NULL,
  [ConfirmStatus] varchar(20) NULL,
  [ConfirmedBy] int NULL,
  [ConfirmedDate] datetime NULL
);

CREATE TABLE [Injection_Result] (
  [ResultID] int IDENTITY(1,1) PRIMARY KEY,
  [FormID] int,
  [StudentID] int,
  [AdministeredBy] int,
  [AdministeredDate] datetime NOT NULL,
  [ImmediateReaction] nvarchar(255),
  [FollowUpRequired] bit DEFAULT (0),
  [FollowUpNotes] nvarchar(255)
);

CREATE TABLE [Health_Check_Form] (
  [FormID] int IDENTITY(1,1) PRIMARY KEY,
  [StudentID] int,
  [ParentID] int,
  [CreatedDate] datetime DEFAULT (getdate()),
  [ConsentStatus] varchar(20) DEFAULT 'Pending',
  [ConsentDate] datetime,
  [GradeLevel] int NULL,
  [ClassName] nvarchar(50) NULL,
  [ConfirmStatus] varchar(20) NULL,
  [ConfirmedBy] int NULL,
  [ConfirmedDate] datetime NULL
);

CREATE TABLE [Health_Check_Result] (
  [ResultID] int IDENTITY(1,1) PRIMARY KEY,
  [FormID] int,
  [StudentID] int,
  [ExaminedBy] int,
  [ExaminedDate] datetime NOT NULL,
  [Height] decimal(5,2),
  [Weight] decimal(5,2),
  [VisionRight] nvarchar(20),
  [VisionLeft] nvarchar(20),
  [HearingStatus] nvarchar(50),
  [BloodPressure] nvarchar(20),
  [HeartRate] int,
  [GeneralFindings] nvarchar(1000),
  [Recommendations] nvarchar(1000),
);

CREATE TABLE [Appointment] (
  [AppointmentID] int IDENTITY(1,1) PRIMARY KEY,
  [StudentID] int,
  [ParentID] int,
  [StaffID] int,
  [AppointmentDate] datetime NOT NULL,
  [AppointmentType] varchar(50) NOT NULL,
  [Reason] nvarchar(255) NOT NULL,
  [Status] varchar(20) DEFAULT 'Scheduled',
  [Notes] nvarchar(500),
  [CreatedBy] int,
  [CreatedDate] datetime DEFAULT (getdate()),
);

CREATE TABLE [Report] (
  [ReportID] int IDENTITY(1,1) PRIMARY KEY,
  [ReportType] varchar(50) NOT NULL,
  [ReportName] nvarchar(100) NOT NULL,
  [GeneratedDate] datetime DEFAULT (getdate()),
  [DateRange_Start] date,
  [DateRange_End] date,
  [InjectionData] nvarchar(max),
  [HealthCheckData] nvarchar(max),
  [HealthEventData] nvarchar(max),
  [InventoryData] nvarchar(max),
  [NonParticipantData] nvarchar(max),
  [AppointmentData] nvarchar(max),
  [GeneratedBy] int,
  [BasedOnDashboardID] int
);

CREATE TABLE [Dashboard_Summary] (
  [SummaryID] int IDENTITY(1,1) PRIMARY KEY,
  [StaffID] int,
  [GeneratedDate] datetime DEFAULT (getdate()),
  [TotalInjectionParticipants] int DEFAULT (0),
  [TotalInjectionNonParticipants] int DEFAULT (0),
  [InjectionParticipationRate] decimal(5,2) DEFAULT (0),
  [TotalHealthCheckParticipants] int DEFAULT (0),
  [TotalHealthCheckNonParticipants] int DEFAULT (0),
  [HealthCheckParticipationRate] decimal(5,2) DEFAULT (0),
  [TotalHealthEvents] int DEFAULT (0),
  [TotalMedicineRequests] int DEFAULT (0),
  [TotalAppointments] int DEFAULT (0),
  [ScheduledAppointments] int DEFAULT (0),
  [CompletedAppointments] int DEFAULT (0),
  [TotalMedicineItems] int DEFAULT (0),
  [TotalSupplyItems] int DEFAULT (0),
  [InjectionFormID] int NULL,
  [HealthCheckFormID] int NULL,
  [HealthEventID] int NULL,
  [MedicineRequestID] int NULL,
  [AppointmentID] int NULL,
  [MedicineID] int NULL,
  [SupplyID] int NULL
);

CREATE TABLE [Role] (
  [RoleID] int IDENTITY(1,1) PRIMARY KEY,
  [RoleName] varchar(50) UNIQUE NOT NULL
);
